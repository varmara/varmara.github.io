Items per order

What is the average number of items per order? Also find the minimum and maximum.

TipSolution 1 
order_items = order_items.withColumn("year", F.year(F.col("shipping_limit_date")))

order_item_counts = (
    order_items
    .groupBy("year", "order_id")
    .count()
)

order_statistics = (
    order_item_counts
    .agg(
        F.avg(F.col("count")).alias("avg_n_items"),
        F.min(F.col("count")).alias("min"),
        F.max(F.col("count")).alias("max")
    )
)

A Histogram of the number of items per order

TipSolution - plotly 
import plotly.express as px
import plotly.graph_objects as go
import plotly.io as pio
import numpy as np
from pyspark.sql.types import StringType

pio.templates.default = "plotly_white+presentation"

order_items = order_items.withColumn("year", F.year(F.col("shipping_limit_date")).cast(StringType()))

order_item_counts = (
    order_items
    .groupBy("year", "order_id")
    .count()
)

order_item_counts_pdf = order_item_counts.toPandas()
# set the number of bins
fig = px.histogram(
    order_item_counts_pdf, 
    x="count", 
    nbins = 21, 
    histnorm="percent", 
    labels={'count':'Items per order'}, 
    color_discrete_sequence=["indianred"],
    facet_col="year",
    # facet_col_wrap=2,
    category_orders={"year": [str(x) for x in range(2016, 2021, 1) if x != 2019]}
    )
fig.update_traces(
    marker_line_color="black", 
    marker_line_width=0.25, 
    texttemplate="%{y:.1f}%"
    )
fig.update_layout(
    bargap=0.2, 
    title = {"text":"Items per Order", "x": 0.5, "xanchor": "center"}
    )
fig.for_each_annotation(lambda a: a.update(text=a.text.split("=")[-1]))
fig.update_xaxes(title_text="")
# fig.update_layout(xaxis={"title": {"text": "Items per order"}})
fig.add_annotation(
    text="Items per order", 
    xref="paper", yref="paper", x=0.5, y=-0.15, showarrow=False, font=dict(size=16),
)
fig.show()
# set bin sizes
fig.update_traces(
    xbins=dict(start=1, end=21, size=2), 
    texttemplate="%{y:.0f}"
    )
fig.show()
# compute bin sizes in advance
max_val = order_item_counts_pdf["count"].max()
counts, bins = np.histogram(order_item_counts_pdf["count"], bins = range(0, max_val + 2, 3))
# shift to bin centers
bins = 0.5 * (bins[:-1] + bins[1:])

fig = px.bar(
    x=bins, y=counts, 
    labels={'x':'Items per order', 'y':'Count'}, 
    title = "Items per Order"
    )
fig.show()
# plotly - not express
fig = go.Figure(
    data=[go.Histogram(x=order_item_counts_pdf["count"], nbinsx=20)]
)
fig.update_layout(bargap=0.2)
fig.show()
